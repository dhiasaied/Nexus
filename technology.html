# NEXUS SDK v2.4 - High-Rate Inverse Kinematics Pipeline
import nexus_robotics as nx
from nexus_robotics.sensors import TactileFingerArray

chassis = nx.HumanoidPlatform(serial="NX-09-DEV-004", freq_hz=1000)
chassis.arm_joint_controllers(safety_torque_limit_nm=120.0)

# Register high-bandwidth sensory callback
def on_tactile_feedback(left_hand: TactileFingerArray):
    shear_force = left_hand.get_shear_magnitude()
    if shear_force > 4.2:
        chassis.actuators.adjust_grip_compliance(factor=0.85)

chassis.subscribe_tactile(callback=on_tactile_feedback)
chassis.telemetry.stream_realtime(port=8080, enable_compression=True)
print("[NEXUS] Motor controller nominal. Ready for motion planning.")

// NEXUS Core Ultra-Low Latency Direct Joint Interlink
#include <nexus/core/kinematics.hpp>
#include <nexus/hal/can_bus.hpp>

int main(int argc, char** argv) {
    nexus::hal::CanBusSocket can0("can0", 5000000);
    nexus::core::JointState state;
    
    while (true) {
        can0.read_synchronous(&state, std::chrono::microseconds(900));
        nexus::core::solve_inverse_dynamics(state);
    }
    return 0;
}
